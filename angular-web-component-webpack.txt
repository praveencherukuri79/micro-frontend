// FILE: apps/remote-widget/src/app/remote-widget/remote-widget.component.ts
// Simple standalone Angular 17 component used as a web component
// ============================================================================
import { Component, EventEmitter, Input, Output } from '@angular/core';

@Component({
  standalone: true,
  selector: 'app-remote-widget',
  template: `
    <section class="remote-widget">
      <h3>{{ title }}</h3>
      <p>{{ description }}</p>
      <button type="button" (click)="handleClick()">Click from Web Component</button>
    </section>
  `,
  styles: [`
    .remote-widget {
      display: block;
      padding: 12px;
      border: 1px solid #ccc;
      border-radius: 4px;
      font-family: system-ui, -apple-system, BlinkMacSystemFont, "Segoe UI", sans-serif;
      background: #fafafa;
    }

    h3 {
      margin: 0 0 8px;
      font-size: 16px;
      font-weight: 600;
    }

    p {
      margin: 0 0 12px;
      font-size: 13px;
      color: #555;
    }

    button {
      font-size: 13px;
      padding: 6px 12px;
      border-radius: 4px;
      border: 1px solid #999;
      background: #f5f5f5;
      cursor: pointer;
    }

    button:hover {
      background: #e0e0e0;
    }
  `]
})
export class RemoteWidgetComponent {
  @Input() title = 'Remote Widget';
  @Input() description = 'Angular 17 web component, single JS bundle';
  @Output() action = new EventEmitter<void>();

  handleClick(): void {
    this.action.emit();
  }
}

// ============================================================================
// FILE: apps/remote-widget/src/main.webcomponent.ts
// Angular 17 entry used for the WC build. This will be bundled into one
// remote-widget.js file along with Angular runtime, zone.js, etc.
// ============================================================================
import 'zone.js';

import { bootstrapApplication } from '@angular/platform-browser';
import { createCustomElement } from '@angular/elements';
import { RemoteWidgetComponent } from './app/remote-widget/remote-widget.component';

bootstrapApplication(RemoteWidgetComponent)
  .then(appRef => {
    const injector = appRef.injector;
    const RemoteElement = createCustomElement(RemoteWidgetComponent, { injector });

    // This is the tag you will use in plain HTML: <remote-widget>
    if (!customElements.get('remote-widget')) {
      customElements.define('remote-widget', RemoteElement);
    }
  })
  .catch(err => console.error(err));

// ============================================================================
